    // OPEN APPLICATIONS

    private void openApp(String name) {

        try {

            Intent intent;

            switch (name) {

                case "Phone":
                    intent = new Intent(Intent.ACTION_DIAL);
                    break;

                case "Messages":
                    intent = new Intent(Intent.ACTION_SENDTO);
                    intent.setData(Uri.parse("smsto:"));
                    break;

                case "Camera":
                    intent = new Intent(
                            android.provider.MediaStore
                                    .ACTION_IMAGE_CAPTURE);
                    break;

                case "Gallery":
                    intent = new Intent(Intent.ACTION_GET_CONTENT);
                    intent.setType("image/*");
                    intent = Intent.createChooser(
                            intent, "Choose Image");
                    break;

                case "Settings":
                    intent = new Intent(
                            android.provider.Settings.ACTION_SETTINGS);
                    break;

                case "Files":
                    intent = new Intent(Intent.ACTION_OPEN_DOCUMENT);
                    intent.setType("*/*");
                    intent.addCategory(Intent.CATEGORY_OPENABLE);
                    intent = Intent.createChooser(
                            intent, "Choose File");
                    break;

                case "Browser":
                    openBrowser();
                    return;

                case "Falcon":
                    openFalcon();
                    return;

                case "CoreX AI":
                    openCoreXAI();
                    return;

                default:
                    return;
            }

            startActivity(intent);

        } catch (ActivityNotFoundException e) {

            Toast.makeText(this,
                    name + " is not available",
                    Toast.LENGTH_SHORT).show();

        } catch (Exception e) {

            Toast.makeText(this,
                    "Unable to open " + name,
                    Toast.LENGTH_SHORT).show();
        }
    }

    // VOICE INPUT

    private void startVoiceRecognition() {

        try {

            Intent intent = new Intent(
                    RecognizerIntent.ACTION_RECOGNIZE_SPEECH);

            intent.putExtra(
                    RecognizerIntent.EXTRA_LANGUAGE_MODEL,
                    RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);

            intent.putExtra(
                    RecognizerIntent.EXTRA_LANGUAGE,
                    Locale.getDefault());

            intent.putExtra(
                    RecognizerIntent.EXTRA_PROMPT,
                    "Speak your command");

            startActivityForResult(intent, 101);

        } catch (Exception e) {

            Toast.makeText(this,
                    "Voice recognition is not available",
                    Toast.LENGTH_LONG).show();
        }
    }

    @Override
    protected void onActivityResult(
            int requestCode, int resultCode, Intent data) {

        super.onActivityResult(requestCode, resultCode, data);

        if (requestCode == 101 &&
                resultCode == RESULT_OK &&
                data != null) {

            ArrayList<String> results =
                    data.getStringArrayListExtra(
                            RecognizerIntent.EXTRA_RESULTS);

            if (results != null && !results.isEmpty()) {
                executeAICommand(results.get(0));
            }
        }
    }

    // COMMAND PROCESSING

    private void executeAICommand(String command) {

        String c = command.toLowerCase(Locale.ROOT);

        if (c.contains("setting")) {
            openApp("Settings");

        } else if (c.contains("phone") ||
                c.contains("call")) {
            openApp("Phone");

        } else if (c.contains("message") ||
                c.contains("sms")) {
            openApp("Messages");

        } else if (c.contains("camera")) {
            openApp("Camera");

        } else if (c.contains("gallery") ||
                c.contains("photo")) {
            openApp("Gallery");

        } else if (c.contains("file")) {
            openApp("Files");

        } else if (c.contains("browser") ||
                c.contains("internet")) {
            openApp("Browser");

        } else if (c.contains("falcon")) {
            openApp("Falcon");

        } else if (c.contains("time") ||
                c.contains("samay")) {

            String time = new SimpleDateFormat(
                    "hh:mm a",
                    Locale.getDefault()).format(new Date());

            showAIResult("Current time: " + time);

        } else if (c.contains("date") ||
                c.contains("tarikh")) {

            String date = new SimpleDateFormat(
                    "dd MMMM yyyy",
                    Locale.getDefault()).format(new Date());

            showAIResult("Today's date: " + date);

        } else if (c.contains("battery")) {

            updateInformation();
            showAIResult("Battery: " + batteryText.getText());

        } else {

            showAIResult(
                    "I could not understand this command.\n\n" +
                    "Try: Open Settings, Open Phone, " +
                    "Open Camera or What is the time?\n\n" +
                    "Online AI integration is not available yet."
            );
        }
    }

    private void showAIResult(String message) {

                .setTitle("CoreX AI")
                .setMessage(message)
                .setPositiveButton("OK", null)
                .show();
    }

    // DATE AND BATTERY

    private void updateInformation() {

        String date = new SimpleDateFormat(
                "EEE, dd MMM yyyy",
                Locale.getDefault()).format(new Date());

        if (dateText != null) {
            dateText.setText(date);
        }

        Intent status = registerReceiver(
                null,
                new android.content.IntentFilter(
                        Intent.ACTION_BATTERY_CHANGED));

        if (status != null && batteryText != null) {

            int level = status.getIntExtra(
                    BatteryManager.EXTRA_LEVEL, -1);

            int scale = status.getIntExtra(
                    BatteryManager.EXTRA_SCALE, -1);

            if (level >= 0 && scale > 0) {

                int percentage = level * 100 / scale;

                batteryText.setText(percentage + "%  ▰");
            }
        }
    }

package com.corex.falcon;

import android.content.Context;
import android.database.Cursor;
import android.net.Uri;
import android.provider.OpenableColumns;

import android.security.keystore.KeyGenParameterSpec;
import android.security.keystore.KeyProperties;

import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.AEADBadTagException;
import javax.crypto.spec.GCMParameterSpec;

import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.RandomAccessFile;

import java.nio.ByteBuffer;
import java.nio.charset.StandardCharsets;

import java.security.KeyStore;
import java.security.SecureRandom;

import java.util.UUID;

public final class FalconCrypto {

    private static final String KEY_ALIAS = "CoreX_Falcon_AES_KEY";
    private static final String ANDROID_KEYSTORE = "AndroidKeyStore";

    private static final String TRANSFORMATION = "AES/GCM/NoPadding";

    private static final byte[] MAGIC = {
            'F', 'X', 'E', '1'
    };

    private static final int IV_LENGTH = 12;
    private static final int TAG_LENGTH_BITS = 128;
    private static final int MAX_FILENAME_LENGTH = 4096;

    private FalconCrypto() {
    }

    private static SecretKey getOrCreateKey() throws Exception {

        KeyStore keyStore = KeyStore.getInstance(ANDROID_KEYSTORE);
        keyStore.load(null);

        if (keyStore.containsAlias(KEY_ALIAS)) {

            KeyStore.Entry entry =
                    keyStore.getEntry(KEY_ALIAS, null);

            if (entry instanceof KeyStore.SecretKeyEntry) {

                return ((KeyStore.SecretKeyEntry) entry).getSecretKey();
            }
        }

        KeyGenerator keyGenerator =
                KeyGenerator.getInstance(
                        KeyProperties.KEY_ALGORITHM_AES,
                        ANDROID_KEYSTORE
                );

        KeyGenParameterSpec specification =
                new KeyGenParameterSpec.Builder(
                        KEY_ALIAS,
                        KeyProperties.PURPOSE_ENCRYPT
                                | KeyProperties.PURPOSE_DECRYPT
                )
                        .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                        .setEncryptionPaddings(
                                KeyProperties.ENCRYPTION_PADDING_NONE
                        )
                        .setKeySize(256)
                        .setRandomizedEncryptionRequired(true)
                        .build();

        keyGenerator.init(specification);

        return keyGenerator.generateKey();
    }

    private static String getDisplayName(
            Context context,
            Uri uri
    ) {

        String name = null;

        try (Cursor cursor = context.getContentResolver()
                .query(uri, null, null, null, null)) {

            if (cursor != null
                    && cursor.moveToFirst()) {

                int index = cursor.getColumnIndex(
                        OpenableColumns.DISPLAY_NAME
                );

                if (index >= 0) {
                    name = cursor.getString(index);
                }
            }

        } catch (Exception ignored) {
        }

        if (name == null || name.trim().isEmpty()) {
            name = "Falcon_File";
        }

        return sanitizeFileName(name);
    }

    private static String sanitizeFileName(String name) {

        if (name == null || name.trim().isEmpty()) {
            return "Falcon_File";
        }

        String cleaned = name.replaceAll(
                "[\\\\/:*?\"<>|\\p{Cntrl}]",
                "_"
        );

        cleaned = cleaned.trim();

        if (cleaned.isEmpty()) {
            return "Falcon_File";
        }

        if (cleaned.length() > 180) {
            cleaned = cleaned.substring(0, 180);
        }

        return cleaned;
    }

    private static byte[] createAuthenticatedHeader(
            String originalName
    ) {

        byte[] nameBytes = originalName.getBytes(
                StandardCharsets.UTF_8
        );

        ByteBuffer buffer = ByteBuffer.allocate(
                MAGIC.length + 4 + nameBytes.length
        );

        buffer.put(MAGIC);
        buffer.putInt(nameBytes.length);
        buffer.put(nameBytes);

        return buffer.array();
    }

    public static File encrypt(
            Context context,
            Uri sourceUri
    ) throws Exception {

        if (sourceUri == null) {
            throw new IllegalArgumentException(
                    "Source file is missing."
            );
        }

        String originalName = getDisplayName(
                context,
                sourceUri
        );

        byte[] authenticatedHeader =
                createAuthenticatedHeader(originalName);

        byte[] iv = new byte[IV_LENGTH];

        SecureRandom secureRandom = new SecureRandom();
        secureRandom.nextBytes(iv);

        Cipher cipher = Cipher.getInstance(TRANSFORMATION);

        cipher.init(
                Cipher.ENCRYPT_MODE,
                getOrCreateKey(),
                new GCMParameterSpec(TAG_LENGTH_BITS, iv)
        );

        // Filename metadata is authenticated by AES-GCM.
        cipher.updateAAD(authenticatedHeader);

        File vaultDirectory =
                FalconVault.getVaultDirectory(context);

        File encryptedFile = new File(
                vaultDirectory,
                "falcon_"
                        + System.currentTimeMillis()
                        + "_"
                        + UUID.randomUUID()
                        + ".fxe"
        );

        boolean success = false;

        try (
                InputStream input =
                        context.getContentResolver()
                                .openInputStream(sourceUri);

                FileOutputStream output =
                        new FileOutputStream(encryptedFile)
        ) {

            if (input == null) {
                throw new Exception(
                        "Unable to open selected file."
                );
            }

            // Header
            output.write(authenticatedHeader);

            // Initialization vector
            output.write(iv);

            byte[] buffer = new byte[8192];

            int bytesRead;

            while ((bytesRead = input.read(buffer)) != -1) {

                byte[] encryptedChunk =
                        cipher.update(buffer, 0, bytesRead);

                if (encryptedChunk != null
                        && encryptedChunk.length > 0) {

                    output.write(encryptedChunk);
                }
            }

            // Writes final encrypted data and GCM authentication tag.
            byte[] finalEncryptedData = cipher.doFinal();

            if (finalEncryptedData.length > 0) {
                output.write(finalEncryptedData);
            }

            output.flush();

            success = true;

            return encryptedFile;

        } finally {

            if (!success && encryptedFile.exists()) {
                encryptedFile.delete();
            }
        }
    }

    public static String getOriginalFileName(
            File encryptedFile
    ) {

        if (encryptedFile == null
                || !encryptedFile.exists()) {

            return "Unknown File";
        }

        try (
                FileInputStream input =
                        new FileInputStream(encryptedFile)
        ) {

            byte[] magic = new byte[MAGIC.length];

            if (input.read(magic) != MAGIC.length) {
                return "Unknown File";
            }

            for (int i = 0; i < MAGIC.length; i++) {

                if (magic[i] != MAGIC[i]) {
                    return "Unknown File";
                }
            }

            byte[] lengthBytes = new byte[4];

            if (input.read(lengthBytes) != 4) {
                return "Unknown File";
            }

            int nameLength = ByteBuffer
                    .wrap(lengthBytes)
                    .getInt();

            if (nameLength <= 0
                    || nameLength > MAX_FILENAME_LENGTH) {

                return "Unknown File";
            }

            byte[] nameBytes = new byte[nameLength];

            int totalRead = 0;

            while (totalRead < nameLength) {

                int count = input.read(
                        nameBytes,
                        totalRead,
                        nameLength - totalRead
                );

                if (count == -1) {
                    return "Unknown File";
                }

                totalRead += count;
            }

            String name = new String(
                    nameBytes,
                    StandardCharsets.UTF_8
            );

            return sanitizeFileName(name);

        } catch (Exception e) {

            return "Unknown File";
        }
    }

    public static void decrypt(
            File encryptedFile,
            File destinationFile
    ) throws Exception {

        if (encryptedFile == null
                || !encryptedFile.exists()) {

            throw new Exception(
                    "Encrypted file does not exist."
            );
        }

        if (destinationFile == null) {
            throw new IllegalArgumentException(
                    "Destination file is missing."
            );
        }

        boolean success = false;

        try (
                FileInputStream input =
                        new FileInputStream(encryptedFile);

                FileOutputStream output =
                        new FileOutputStream(destinationFile)
        ) {

            // Read and verify file signature.
            byte[] magic = new byte[MAGIC.length];

            if (input.read(magic) != MAGIC.length) {
                throw new Exception(
                        "Invalid Falcon encrypted file."
                );
            }

            for (int i = 0; i < MAGIC.length; i++) {

                if (magic[i] != MAGIC[i]) {
                    throw new Exception(
                            "Unsupported Falcon file format."
                    );
                }
            }

            // Read filename length.
            byte[] lengthBytes = new byte[4];

            if (input.read(lengthBytes) != 4) {
                throw new Exception(
                        "Encrypted file header is incomplete."
                );
            }

            int nameLength = ByteBuffer
                    .wrap(lengthBytes)
                    .getInt();

            if (nameLength <= 0
                    || nameLength > MAX_FILENAME_LENGTH) {

                throw new Exception(
                        "Invalid filename metadata."
                );
            }

            // Read original filename metadata.
            byte[] nameBytes = new byte[nameLength];

            int totalRead = 0;

            while (totalRead < nameLength) {

                int count = input.read(
                        nameBytes,
                        totalRead,
                        nameLength - totalRead
                );

                if (count == -1) {
                    throw new Exception(
                            "Incomplete filename metadata."
                    );
                }

                totalRead += count;
            }

            // Reconstruct the exact authenticated header.
            byte[] authenticatedHeader =
                    new byte[
                            MAGIC.length
                                    + 4
                                    + nameBytes.length
                    ];

            System.arraycopy(
                    MAGIC,
                    0,
                    authenticatedHeader,
                    0,
                    MAGIC.length
            );

            System.arraycopy(
                    lengthBytes,
                    0,
                    authenticatedHeader,
                    MAGIC.length,
                    4
            );

            System.arraycopy(
                    nameBytes,
                    0,
                    authenticatedHeader,
                    MAGIC.length + 4,
                    nameBytes.length
            );

            // Read initialization vector.
            byte[] iv = new byte[IV_LENGTH];

            if (input.read(iv) != IV_LENGTH) {
                throw new Exception(
                        "Encryption IV is missing."
                );
            }

            Cipher cipher = Cipher.getInstance(TRANSFORMATION);

            cipher.init(
                    Cipher.DECRYPT_MODE,
                    getOrCreateKey(),
                    new GCMParameterSpec(TAG_LENGTH_BITS, iv)
            );

            // Verify filename metadata during GCM authentication.
            cipher.updateAAD(authenticatedHeader);

            byte[] buffer = new byte[8192];

            int bytesRead;

            while ((bytesRead = input.read(buffer)) != -1) {

                byte[] decryptedChunk =
                        cipher.update(buffer, 0, bytesRead);

                if (decryptedChunk != null
                        && decryptedChunk.length > 0) {

                    output.write(decryptedChunk);
                }
            }

            // Authentication tag is verified here.
            // Tampered or corrupted files will fail.
            byte[] finalDecryptedData = cipher.doFinal();

            if (finalDecryptedData.length > 0) {
                output.write(finalDecryptedData);
            }

            output.flush();

            success = true;

        } catch (Exception e) {

            if (e instanceof AEADBadTagException) {

                throw new SecurityException(
                        "Falcon security verification failed. "
                                + "File may have been modified.",
                        e
                );
            }

            throw e;

        } finally {

            if (!success && destinationFile.exists()) {
                destinationFile.delete();
            }
        }
    }

package com.corex.falcon;

import android.security.keystore.KeyGenParameterSpec;
import android.security.keystore.KeyProperties;

import android.util.Base64;

import java.nio.charset.StandardCharsets;

import java.security.KeyFactory;
import java.security.KeyPair;
import java.security.KeyPairGenerator;
import java.security.KeyStore;
import java.security.MessageDigest;
import java.security.PublicKey;

import java.security.spec.MGF1ParameterSpec;
import java.security.spec.X509EncodedKeySpec;

import javax.crypto.Cipher;
import javax.crypto.spec.OAEPParameterSpec;
import javax.crypto.spec.PSource;

public final class FalconIdentity {

    private static final String KEY_ALIAS =
            "CoreX_Falcon_Identity_RSA";

    private static final String KEYSTORE =
            "AndroidKeyStore";

    private FalconIdentity() {
    }

    private static KeyPair getOrCreateIdentity() throws Exception {

        KeyStore keyStore = KeyStore.getInstance(KEYSTORE);
        keyStore.load(null);

        if (keyStore.containsAlias(KEY_ALIAS)) {

            KeyStore.PrivateKeyEntry entry =
                    (KeyStore.PrivateKeyEntry)
                            keyStore.getEntry(KEY_ALIAS, null);

            return new KeyPair(
                    entry.getCertificate().getPublicKey(),
                    entry.getPrivateKey()
            );
        }

        KeyPairGenerator generator =
                KeyPairGenerator.getInstance(
                        KeyProperties.KEY_ALGORITHM_RSA,
                        KEYSTORE
                );

        KeyGenParameterSpec specification =
                new KeyGenParameterSpec.Builder(
                        KEY_ALIAS,
                        KeyProperties.PURPOSE_ENCRYPT
                                | KeyProperties.PURPOSE_DECRYPT
                )
                        .setKeySize(3072)
                        .setDigests(
                                KeyProperties.DIGEST_SHA256,
                                KeyProperties.DIGEST_SHA512
                        )
                        .setEncryptionPaddings(
                                KeyProperties.ENCRYPTION_PADDING_RSA_OAEP
                        )
                        .setRandomizedEncryptionRequired(true)
                        .build();

        generator.initialize(specification);

        return generator.generateKeyPair();
    }

    public static String getPublicKey() throws Exception {

        PublicKey publicKey =
                getOrCreateIdentity().getPublic();

        return Base64.encodeToString(
                publicKey.getEncoded(),
                Base64.NO_WRAP
        );
    }

    public static String getFingerprint() throws Exception {

        byte[] publicKey =
                getOrCreateIdentity()
                        .getPublic()
                        .getEncoded();

        byte[] digest = MessageDigest
                .getInstance("SHA-256")
                .digest(publicKey);

        StringBuilder result = new StringBuilder();

        for (byte value : digest) {

            result.append(
                    String.format("%02X", value & 0xFF)
            );
        }

        return result.toString();
    }

    public static byte[] encryptForRecipient(
            byte[] data,
            String recipientPublicKey
    ) throws Exception {

        if (data == null || data.length == 0) {
            throw new IllegalArgumentException(
                    "Encryption data is empty."
            );
        }

        if (recipientPublicKey == null
                || recipientPublicKey.trim().isEmpty()) {

            throw new IllegalArgumentException(
                    "Recipient public key is missing."
            );
        }

        byte[] decodedKey = Base64.decode(
                recipientPublicKey,
                Base64.DEFAULT
        );

        PublicKey publicKey = KeyFactory
                .getInstance("RSA")
                .generatePublic(
                        new X509EncodedKeySpec(decodedKey)
                );

        Cipher cipher = Cipher.getInstance(
                "RSA/ECB/OAEPWithSHA-256AndMGF1Padding"
        );

        OAEPParameterSpec parameters =
                new OAEPParameterSpec(
                        "SHA-256",
                        "MGF1",
                        MGF1ParameterSpec.SHA1,
                        PSource.PSpecified.DEFAULT
                );

        cipher.init(
                Cipher.ENCRYPT_MODE,
                publicKey,
                parameters
        );

        return cipher.doFinal(data);
    }

    public static byte[] decryptForSelf(
            byte[] encryptedData
    ) throws Exception {

        if (encryptedData == null
                || encryptedData.length == 0) {

            throw new IllegalArgumentException(
                    "Encrypted data is empty."
            );
        }

        Cipher cipher = Cipher.getInstance(
                "RSA/ECB/OAEPWithSHA-256AndMGF1Padding"
        );

        OAEPParameterSpec parameters =
                new OAEPParameterSpec(
                        "SHA-256",
                        "MGF1",
                        MGF1ParameterSpec.SHA1,
                        PSource.PSpecified.DEFAULT
                );

        cipher.init(
                Cipher.DECRYPT_MODE,
                getOrCreateIdentity().getPrivate(),
                parameters
        );

        return cipher.doFinal(encryptedData);
    }

    public static boolean verifyFingerprint(
            String publicKeyBase64,
            String expectedFingerprint
    ) throws Exception {

        if (publicKeyBase64 == null
                || expectedFingerprint == null) {

            return false;
        }

        byte[] decodedKey = Base64.decode(
                publicKeyBase64,
                Base64.DEFAULT
        );

        byte[] digest = MessageDigest
                .getInstance("SHA-256")
                .digest(decodedKey);

        StringBuilder actual = new StringBuilder();

        for (byte value : digest) {

            actual.append(
                    String.format("%02X", value & 0xFF)
            );
        }

        return MessageDigest.isEqual(
                actual.toString().getBytes(StandardCharsets.UTF_8),
                expectedFingerprint
                        .replace(" ", "")
                        .toUpperCase()
                        .getBytes(StandardCharsets.UTF_8)
        );
    }

package com.corex.falcon;

import javax.crypto.Cipher;
import javax.crypto.KeyGenerator;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;

import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.OutputStream;

import java.nio.ByteBuffer;

import java.security.SecureRandom;

import java.util.Arrays;

public final class FalconTransferCrypto {

    private static final byte[] MAGIC = {
            'F', 'T', 'X', '1'
    };

    private static final int AES_KEY_SIZE = 256;
    private static final int IV_LENGTH = 12;
    private static final int TAG_LENGTH_BITS = 128;

    private static final int MAX_WRAPPED_KEY_LENGTH = 1024;

    private static final String TRANSFORMATION =
            "AES/GCM/NoPadding";

    private FalconTransferCrypto() {
    }

    private static SecretKey generateAESKey() throws Exception {

        KeyGenerator generator =
                KeyGenerator.getInstance("AES");

        generator.init(AES_KEY_SIZE);

        return generator.generateKey();
    }

    private static byte[] generateIV() {

        byte[] iv = new byte[IV_LENGTH];

        new SecureRandom().nextBytes(iv);

        return iv;
    }

    /**
     * Encrypts a file for a recipient.
     *
     * The recipient's RSA public key protects
     * the randomly generated AES key.
     */
    public static void encryptForRecipient(
            File sourceFile,
            File destinationFile,
            String recipientPublicKey
    ) throws Exception {

        if (sourceFile == null
                || !sourceFile.isFile()
                || !sourceFile.canRead()) {

            throw new Exception(
                    "Source file is unavailable."
            );
        }

        if (destinationFile == null) {
            throw new IllegalArgumentException(
                    "Destination file is missing."
            );
        }

        if (recipientPublicKey == null
                || recipientPublicKey.trim().isEmpty()) {

            throw new IllegalArgumentException(
                    "Recipient public key is missing."
            );
        }

        SecretKey aesKey = generateAESKey();

        byte[] wrappedKey =
                FalconIdentity.encryptForRecipient(
                        aesKey.getEncoded(),
                        recipientPublicKey
                );

        if (wrappedKey.length == 0
                || wrappedKey.length > MAX_WRAPPED_KEY_LENGTH) {

            throw new Exception(
                    "Invalid encrypted AES key."
            );
        }

        byte[] iv = generateIV();

        ByteBuffer headerBuffer = ByteBuffer.allocate(
                MAGIC.length
                        + 4
                        + wrappedKey.length
                        + IV_LENGTH
        );

        headerBuffer.put(MAGIC);
        headerBuffer.putInt(wrappedKey.length);
        headerBuffer.put(wrappedKey);
        headerBuffer.put(iv);

        byte[] authenticatedHeader =
                headerBuffer.array();

        Cipher cipher = Cipher.getInstance(TRANSFORMATION);

        cipher.init(
                Cipher.ENCRYPT_MODE,
                aesKey,
                new GCMParameterSpec(
                        TAG_LENGTH_BITS,
                        iv
                )
        );

        cipher.updateAAD(authenticatedHeader);

        boolean success = false;

        try (
                InputStream input =
                        new FileInputStream(sourceFile);

                OutputStream output =
                        new FileOutputStream(destinationFile)
        ) {

            output.write(authenticatedHeader);

            byte[] buffer = new byte[8192];

            int bytesRead;

            while ((bytesRead = input.read(buffer)) != -1) {

                byte[] encrypted =
                        cipher.update(
                                buffer,
                                0,
                                bytesRead
                        );

                if (encrypted != null
                        && encrypted.length > 0) {

                    output.write(encrypted);
                }
            }

            byte[] finalData = cipher.doFinal();

            if (finalData.length > 0) {
                output.write(finalData);
            }

            output.flush();

            success = true;

        } finally {

            if (!success && destinationFile.exists()) {
                destinationFile.delete();
            }
        }
    }

    /**
     * Decrypts a file addressed to this device's
     * Falcon RSA identity.
     */
    public static void decryptReceivedFile(
            File encryptedFile,
            File destinationFile
    ) throws Exception {

        if (encryptedFile == null
                || !encryptedFile.isFile()
                || !encryptedFile.canRead()) {

            throw new Exception(
                    "Encrypted file is unavailable."
            );
        }

        if (destinationFile == null) {
            throw new IllegalArgumentException(
                    "Destination file is missing."
            );
        }

        boolean success = false;

        try (
                InputStream input =
                        new FileInputStream(encryptedFile);

                OutputStream output =
                        new FileOutputStream(destinationFile)
        ) {

            // Read file signature.
            byte[] magic = new byte[MAGIC.length];

            readFully(input, magic);

            if (!Arrays.equals(magic, MAGIC)) {

                throw new Exception(
                        "Invalid Falcon transfer file."
                );
            }

            // Read wrapped AES key length.
            byte[] lengthBytes = new byte[4];

            readFully(input, lengthBytes);

            int wrappedKeyLength =
                    ByteBuffer.wrap(lengthBytes).getInt();

            if (wrappedKeyLength <= 0
                    || wrappedKeyLength > MAX_WRAPPED_KEY_LENGTH) {

                throw new Exception(
                        "Invalid wrapped AES key."
                );
            }

            // Read RSA encrypted AES key.
            byte[] wrappedKey =
                    new byte[wrappedKeyLength];

            readFully(input, wrappedKey);

            // Read initialization vector.
            byte[] iv = new byte[IV_LENGTH];

            readFully(input, iv);

            // Reconstruct authenticated header.
            ByteBuffer headerBuffer = ByteBuffer.allocate(
                    MAGIC.length
                            + 4
                            + wrappedKey.length
                            + IV_LENGTH
            );

            headerBuffer.put(MAGIC);
            headerBuffer.putInt(wrappedKeyLength);
            headerBuffer.put(wrappedKey);
            headerBuffer.put(iv);

            byte[] authenticatedHeader =
                    headerBuffer.array();

            // Recover AES key using this device's RSA identity.
            byte[] rawAESKey =
                    FalconIdentity.decryptForSelf(wrappedKey);

            if (rawAESKey.length != 32) {

                throw new SecurityException(
                        "Invalid AES-256 key."
                );
            }

            SecretKey aesKey = new SecretKeySpec(
                    rawAESKey,
                    "AES"
            );

            Cipher cipher = Cipher.getInstance(TRANSFORMATION);

            cipher.init(
                    Cipher.DECRYPT_MODE,
                    aesKey,
                    new GCMParameterSpec(
                            TAG_LENGTH_BITS,
                            iv
                    )
            );

            cipher.updateAAD(authenticatedHeader);

            byte[] buffer = new byte[8192];

            int bytesRead;

            while ((bytesRead = input.read(buffer)) != -1) {

                byte[] decrypted =
                        cipher.update(
                                buffer,
                                0,
                                bytesRead
                        );

                if (decrypted != null
                        && decrypted.length > 0) {

                    output.write(decrypted);
                }
            }

            // GCM authentication is verified here.
            byte[] finalData = cipher.doFinal();

            if (finalData.length > 0) {
                output.write(finalData);
            }

            output.flush();

            success = true;

        } finally {

            if (!success && destinationFile.exists()) {
                destinationFile.delete();
            }
        }
    }

    private static void readFully(
            InputStream input,
            byte[] data
    ) throws Exception {

        int offset = 0;

        while (offset < data.length) {

            int count = input.read(
                    data,
                    offset,
                    data.length - offset
            );

            if (count == -1) {

                throw new Exception(
                        "Incomplete Falcon transfer file."
                );
            }

            offset += count;
        }
    }

package com.corex.falcon;

import android.content.Context;

import java.io.File;
import java.util.Arrays;
import java.util.Comparator;

public final class FalconVault {

    private FalconVault() {
    }

    public static File getVaultDirectory(Context context) throws Exception {

        File directory = new File(
                context.getFilesDir(),
                "falcon_vault"
        );

        if (!directory.exists() && !directory.mkdirs()) {
            throw new Exception("Falcon Vault could not be created.");
        }

        return directory.getCanonicalFile();
    }

    public static File getRecoveryDirectory(Context context) throws Exception {

        File directory = new File(
                context.getFilesDir(),
                "falcon_recovered"
        );

        if (!directory.exists() && !directory.mkdirs()) {
            throw new Exception("Recovery directory could not be created.");
        }

        return directory.getCanonicalFile();
    }

    public static File[] getEncryptedFiles(Context context) throws Exception {

        File[] files = getVaultDirectory(context).listFiles(
                (dir, name) -> name.endsWith(".fxe")
        );

        if (files == null) {
            return new File[0];
        }

        Arrays.sort(
                files,
                Comparator.comparingLong(File::lastModified).reversed()
        );

        return files;
    }

    public static File recover(
            Context context,
            File encryptedFile
    ) throws Exception {

        if (encryptedFile == null || !encryptedFile.exists()) {
            throw new Exception("Encrypted file does not exist.");
        }

        File vault = getVaultDirectory(context);

        if (!encryptedFile.getCanonicalFile().getParentFile()
                .equals(vault)) {
            throw new SecurityException("File is outside Falcon Vault.");
        }

        File recoveredFile = new File(
                getRecoveryDirectory(context),
                "recovered_" + System.currentTimeMillis() + ".bin"
        );

        try {

            FalconCrypto.decrypt(encryptedFile, recoveredFile);

            return recoveredFile;

        } catch (Exception e) {

            if (recoveredFile.exists()) {
                recoveredFile.delete();
            }

            throw e;
        }
    }

    public static boolean deleteEncryptedFile(
            Context context,
            File encryptedFile
    ) throws Exception {

        if (encryptedFile == null || !encryptedFile.exists()) {
            return false;
        }

        File vault = getVaultDirectory(context);

        if (!encryptedFile.getCanonicalFile().getParentFile()
                .equals(vault)) {
            throw new SecurityException("Invalid Falcon Vault file.");
        }

        if (!encryptedFile.getName().endsWith(".fxe")) {
            throw new SecurityException("Invalid encrypted file type.");
        }

        return encryptedFile.delete();
    }

    @Override
    protected void onActivityResult(
            int requestCode,
            int resultCode,
            Intent data
    ) {

        super.onActivityResult(requestCode, resultCode, data);

        if (requestCode != PICK_FILES ||
                resultCode != RESULT_OK ||
                data == null) {
            return;
        }

        if (isEncrypting) {
            showToast("Please wait for encryption to finish.");
            return;
        }

        if (data.getClipData() != null) {

            int count = data.getClipData().getItemCount();

            for (int i = 0; i < count; i++) {

                Uri uri = data.getClipData()
                        .getItemAt(i)
                        .getUri();

                addAttachment(uri);
            }

        } else if (data.getData() != null) {

            addAttachment(data.getData());
        }

        encryptedFiles.clear();

        encryptionStatus.setText(
                "Files selected. Encryption pending."
        );

        updateAttachmentList();
    }

    private void addAttachment(Uri uri) {

        if (attachments.contains(uri)) {
            return;
        }

        try {

            getContentResolver().takePersistableUriPermission(
                    uri,
                    Intent.FLAG_GRANT_READ_URI_PERMISSION
            );

        } catch (Exception ignored) {
        }

        attachments.add(uri);

        String name = uri.getLastPathSegment();

        if (name == null || name.isEmpty()) {
            name = "Selected file";
        }

        attachmentNames.add(name);
    }

    private void updateAttachmentList() {

        if (attachments.isEmpty()) {
            attachmentList.setText("No files selected.");
            return;
        }

        StringBuilder builder = new StringBuilder();

        builder.append("Selected files: ")
                .append(attachments.size())
                .append("\n\n");

        for (int i = 0; i < attachmentNames.size(); i++) {

            builder.append(i + 1)
                    .append(". ")
                    .append(attachmentNames.get(i))
                    .append("\n");
        }

        attachmentList.setText(builder.toString());
    }

    private void startVoiceRecording() {

        if (checkSelfPermission(Manifest.permission.RECORD_AUDIO)
                != PackageManager.PERMISSION_GRANTED) {

            requestPermissions(
                    new String[]{Manifest.permission.RECORD_AUDIO},
                    RECORD_PERMISSION
            );

            return;
        }

        try {

            File directory = new File(
                    getExternalFilesDir(null),
                    "Falcon"
            );

            if (!directory.exists() && !directory.mkdirs()) {
                throw new Exception("Cannot create recording folder");
            }

            recordingFile = new File(
                    directory,
                    "Voice_" + System.currentTimeMillis() + ".m4a"
            );

            recorder = new MediaRecorder();

            recorder.setAudioSource(MediaRecorder.AudioSource.MIC);
            recorder.setOutputFormat(MediaRecorder.OutputFormat.MPEG_4);
            recorder.setAudioEncoder(MediaRecorder.AudioEncoder.AAC);
            recorder.setOutputFile(recordingFile.getAbsolutePath());

            recorder.prepare();
            recorder.start();

            isRecording = true;

            recordButton.setText("Stop Voice Recording");
            recordingStatus.setText("Recording in progress...");

        } catch (Exception e) {

            releaseRecorder();

            showToast("Recording could not start.");
        }
    }

    private void stopVoiceRecording() {

        if (recorder == null) {
            return;
        }

        try {

            recorder.stop();

            Uri audioUri = Uri.fromFile(recordingFile);

            attachments.add(audioUri);
            attachmentNames.add(recordingFile.getName());

            updateAttachmentList();

            encryptionStatus.setText(
                    "Voice recording saved. Encryption pending."
            );

            recordingStatus.setText("Voice recording saved.");

            showToast("Voice recording saved.");

        } catch (Exception e) {

            if (recordingFile != null) {
                recordingFile.delete();
            }

            recordingStatus.setText("Recording failed.");

        } finally {

            releaseRecorder();

            isRecording = false;

            recordButton.setText("Start Voice Recording");
        }
    }

    private void releaseRecorder() {

        if (recorder != null) {

            try {
                recorder.reset();
            } catch (Exception ignored) {
            }

            try {
                recorder.release();
            } catch (Exception ignored) {
            }

            recorder = null;
        }
    }

    private void prepareTransfer() {

        if (isEncrypting) {
            return;
        }

        String recipient = recipientInput.getText()
                .toString()
                .trim();

        String message = messageInput.getText()
                .toString()
                .trim();

        if (recipient.isEmpty()) {

            recipientInput.setError("Enter recipient Falcon ID");
            recipientInput.requestFocus();
            return;
        }

        if (message.isEmpty() && attachments.isEmpty()) {

            showToast("Write a message or select a file.");
            return;
        }

        if (attachments.isEmpty()) {

            new AlertDialog.Builder(this)
                    .setTitle("Message Prepared")
                    .setMessage(
                            "Recipient: " + recipient +
                            "\n\nText is ready, but network transfer " +
                            "is not configured."
                    )
                    .setPositiveButton("OK", null)
                    .show();

            return;
        }

        isEncrypting = true;
        prepareButton.setEnabled(false);

        encryptionStatus.setText("Encrypting files...");

        ArrayList<Uri> filesToEncrypt =
                new ArrayList<>(attachments);

        executor.execute(() -> {

            ArrayList<File> results = new ArrayList<>();
            String error = null;

            try {

                for (Uri uri : filesToEncrypt) {

                    File encrypted = FalconCrypto.encrypt(
                            getApplicationContext(),
                            uri
                    );

                    results.add(encrypted);
                }

            } catch (Exception e) {

                error = e.getMessage();

                for (File file : results) {
                    file.delete();
                }
            }

            final String finalError = error;

            runOnUiThread(() -> {

                isEncrypting = false;
                prepareButton.setEnabled(true);

                if (finalError != null) {

                    encryptionStatus.setText("Encryption failed.");

                    new AlertDialog.Builder(this)
                            .setTitle("Encryption Failed")
                            .setMessage(
                                    finalError == null
                                            ? "Unknown error."
                                            : finalError
                            )
                            .setPositiveButton("OK", null)
                            .show();

                    return;
                }

                encryptedFiles.clear();
                encryptedFiles.addAll(results);

                encryptionStatus.setText(
                        encryptedFiles.size() +
                        " file(s) encrypted successfully."
                );

                new AlertDialog.Builder(this)
                        .setTitle("Local Encryption Complete")
                        .setMessage(
                                "Recipient: " + recipient +
                                "\nAttachments encrypted: " +
                                encryptedFiles.size() +
                                "\n\nEncrypted copies are stored in " +
                                "Falcon's private app storage.\n\n" +
                                "They have NOT been sent to the recipient. " +
                                "Network transfer and device-to-device " +
                                "encryption are not implemented."
                        )
                        .setPositiveButton("OK", null)
                        .show();
            });
        });
    }

    @Override
    protected void onDestroy() {

        if (isRecording) {

            try {
                recorder.stop();
            } catch (Exception ignored) {
            }

            releaseRecorder();
        }

        executor.shutdownNow();

        super.onDestroy();
    }

    @Override
    protected void onActivityResult(
            int requestCode,
            int resultCode,
            Intent data
    ) {

        super.onActivityResult(
                requestCode,
                resultCode,
                data
        );

        if (requestCode != CREATE_RECOVERED_FILE ||
                resultCode != RESULT_OK ||
                data == null ||
                data.getData() == null) {

            return;
        }

        if (selectedEncryptedFile == null) {
            return;
        }

        Uri destination = data.getData();

        File encryptedFile = selectedEncryptedFile;

        operationRunning = true;

        status.setText("Decrypting file...");

        executor.execute(() -> {

            File temporaryFile = null;

            String error = null;

            try {

                temporaryFile = new File(
                        getCacheDir(),
                        "recovery_" + System.nanoTime() + ".tmp"
                );

                FalconCrypto.decrypt(
                        encryptedFile,
                        temporaryFile
                );

                try (
                        InputStream input =
                                new FileInputStream(temporaryFile);

                        OutputStream output =
                                getContentResolver()
                                        .openOutputStream(destination, "w")
                ) {

                    if (output == null) {
                        throw new Exception(
                                "Cannot open destination."
                        );
                    }

                    byte[] buffer = new byte[8192];

                    int count;

                    while ((count = input.read(buffer)) != -1) {

                        output.write(buffer, 0, count);
                    }

                    output.flush();
                }

            } catch (Exception e) {

                error = e.getMessage();

            } finally {

                if (temporaryFile != null &&
                        temporaryFile.exists()) {

                    temporaryFile.delete();
                }
            }

            final String finalError = error;

            runOnUiThread(() -> {

                operationRunning = false;

                if (finalError == null) {

                    status.setText("Recovery completed.");

                    showToast("File recovered successfully.");

                } else {

                    status.setText("Recovery failed.");

                    new AlertDialog.Builder(this)
                            .setTitle("Recovery Failed")
                            .setMessage(
                                    "File could not be recovered. " +
                                    "It may be damaged or the key unavailable."
                            )
                            .setPositiveButton("OK", null)
                            .show();
                }
            });
        });
    }
